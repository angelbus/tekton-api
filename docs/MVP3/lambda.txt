Instead of baking the corporate CA bundle into the Lambda package or a Lambda Layer, storing it in a dedicated configuration bucket has several advantages:

1) Single source of truth for certificates.
2) Certificate rotation without redeploying Lambdas.
3) Environment-specific bundles (dev, pre, pro).
4) Reusable by every XENA Lambda.
5) Fits nicely with the existing xena-config-{env} concept.

Since Lambda execution environments are reused, XENA can:

- Download it only once during cold start.
- Save it under /tmp.
- Reuse it for all warm invocations.

The /tmp directory persists for the lifetime of the execution environment, making it ideal for this purpose.

Bucket structure:
xena-config-dev/
    certificates/
        corporate-ca.pem

xena-config-pre/
    certificates/
        corporate-ca.pem

xena-config-pro/
    certificates/
        corporate-ca.pem

import os

CA_BUNDLE_PATH = None

def initialize():
    global CA_BUNDLE_PATH

    CA_BUNDLE_PATH = ensure_ca_bundle()

import logging
import os
from pathlib import Path

import boto3
from botocore.exceptions import ClientError

logger = logging.getLogger(__name__)

s3_client = boto3.client("s3")


def ensure_ca_bundle() -> str:
    """
    Downloads the corporate CA bundle from the XENA configuration bucket
    during Lambda cold start.

    The certificate is downloaded only once and cached in /tmp for the
    lifetime of the Lambda execution environment.

    Returns:
        Local filesystem path to the CA bundle.

    Raises:
        RuntimeError if the certificate cannot be downloaded.
    """

    env = os.environ["ENVIRONMENT"]

    bucket = f"xena-config-{env}"
    key = "certificates/corporate-ca.pem"

    local_path = "/tmp/corporate-ca.pem"

    # Already downloaded in this execution environment.
    if Path(local_path).exists():
        logger.info("Using cached corporate CA bundle: %s", local_path)
        return local_path

    logger.info(
        "Downloading corporate CA bundle from s3://%s/%s",
        bucket,
        key,
    )

    try:
        s3_client.download_file(
            Bucket=bucket,
            Key=key,
            Filename=local_path,
        )

    except ClientError as ex:
        logger.exception("Unable to download corporate CA bundle")
        raise RuntimeError(
            f"Unable to download CA bundle from s3://{bucket}/{key}"
        ) from ex

    logger.info("Corporate CA bundle downloaded successfully.")

    return local_path

----

_CA_BUNDLE = None


def get_ca_bundle_path() -> str:
    global _CA_BUNDLE

    if _CA_BUNDLE is None:
        _CA_BUNDLE = ensure_ca_bundle()

    return _CA_BUNDLE
